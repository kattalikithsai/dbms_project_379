How Grievance Redressal Works

Every submission receives an immutable tracking code and moves through strict departmental Service Level Agreements (SLAs).

01

Lodge Grievance

Provide location, photos, category, and description. System issues a unique ID e.g., GRV-2026-XXXX.

02

Official Assignment

Automatically routed to the respective Department Head (PWD, Water Board, Electricity, etc.) and assigned to a field engineer.

03

Field Action

Department conducts site inspection and ground repairs. Live status updates and progress notes are logged in real-time.

04

Verified Resolution

Action notes posted, case marked Resolved, and citizen receives digital completion acknowledgement with audit certificate.

Supabase PostgreSQL Schema Setup

Instructions to run on Supabase:
1. Go to your Supabase Project: Supabase SQL Editor (kihkzjzgpuenbetlrsxp)
2. Paste the SQL script below and click Run.
3. Once executed, click "Check Connection Again" below to sync with live Supabase PostgreSQL!
schema.sql 
-- SUPABASE POSTGRESQL DDL SCHEMA
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

DROP TABLE IF EXISTS grievance_logs CASCADE;
DROP TABLE IF EXISTS grievances CASCADE;
DROP TABLE IF EXISTS profiles CASCADE;
DROP TABLE IF EXISTS departments CASCADE;

CREATE TABLE departments (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    code TEXT NOT NULL UNIQUE,
    description TEXT,
    head_official TEXT,
    icon TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE profiles (
    id TEXT PRIMARY KEY,
    email TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    full_name TEXT NOT NULL,
    phone TEXT,
    role TEXT NOT NULL CHECK (role IN ('citizen', 'official', 'admin')),
    department_id TEXT REFERENCES departments(id) ON DELETE SET NULL,
    district TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE grievances (
    id TEXT PRIMARY KEY,
    tracking_id TEXT UNIQUE NOT NULL,
    citizen_id TEXT REFERENCES profiles(id) ON DELETE SET NULL,
    citizen_name TEXT NOT NULL,
    citizen_email TEXT NOT NULL,
    citizen_phone TEXT,
    title TEXT NOT NULL,
    department_id TEXT REFERENCES departments(id) ON DELETE RESTRICT,
    department_name TEXT NOT NULL,
    priority TEXT NOT NULL CHECK (priority IN ('Low', 'Medium', 'High', 'Urgent')),
    location TEXT NOT NULL,
    district TEXT NOT NULL,
    description TEXT NOT NULL,
    attachment_url TEXT,
    status TEXT NOT NULL DEFAULT 'Pending' CHECK (status IN ('Pending', 'In Progress', 'Resolved', 'Rejected')),
    assigned_official_id TEXT REFERENCES profiles(id) ON DELETE SET NULL,
    assigned_official_name TEXT,
    resolution_notes TEXT,
    resolved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE grievance_logs (
    id TEXT PRIMARY KEY,
    grievance_id TEXT REFERENCES grievances(id) ON DELETE CASCADE,
    tracking_id TEXT NOT NULL,
    action TEXT NOT NULL,
    previous_status TEXT,
    new_status TEXT,
    performed_by_name TEXT NOT NULL,
    performed_by_role TEXT NOT NULL,
    remarks TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE departments ENABLE ROW LEVEL SECURITY;
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE grievances ENABLE ROW LEVEL SECURITY;
ALTER TABLE grievance_logs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public all on departments" ON departments FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public all on profiles" ON profiles FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public all on grievances" ON grievances FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public all on grievance_logs" ON grievance_logs FOR ALL USING (true) WITH CHECK (true);
